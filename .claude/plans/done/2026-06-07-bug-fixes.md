```typescript
import { getRequestConfig } from 'next-intl/server'
import { cookies } from 'next/headers'
import fs from 'fs'
import path from 'path'

const SUPPORTED_LOCALES = ['ja', 'en'] as const
type SupportedLocale = (typeof SUPPORTED_LOCALES)[number]

const DEFAULT_LOCALE: SupportedLocale = 'ja'
const LOCALE_COOKIE = 'NEXT_LOCALE'

const isSupportedLocale = (locale: string): locale is SupportedLocale =>
  (SUPPORTED_LOCALES as readonly string[]).includes(locale)

const readLocaleFile = (locale: string, namespace: string): Record<string, unknown> => {
  const filePath = path.join(process.cwd(), 'public', 'locales', locale, `${namespace}.json`)
  return JSON.parse(fs.readFileSync(filePath, 'utf-8')) as Record<string, unknown>
}

export default getRequestConfig(async ({ requestLocale }) => {
  const requested = await requestLocale
  // middleware が未設定のため requestLocale は常に undefined。
  // Cookie NEXT_LOCALE をフォールバックとして使う。
  const cookieLocale = (await cookies()).get(LOCALE_COOKIE)?.value
  const locale =
    (requested && isSupportedLocale(requested)) ? requested :
    (cookieLocale && isSupportedLocale(cookieLocale)) ? cookieLocale :
    DEFAULT_LOCALE

  const [
    scanMessages,
    commonMessages,
    settingsMessages,
    onboardingMessages,
    historyMessages,
    authMessages,
  ] = ['scan', 'common', 'settings', 'onboarding', 'history', 'auth'].map((ns) =>
    readLocaleFile(locale, ns),
  )

  return {
    locale,
    messages: {
      scan: scanMessages,
      common: commonMessages,
      settings: settingsMessages,
      onboarding: onboardingMessages,
      history: historyMessages,
      auth: authMessages,
    },
  }
})
```

## Task 4: 引継ぎ機能バックエンド実装（Bug 3）

**Files:**
- Modify: `backend/prisma/schema.prisma`（`BackupCode` モデル追加）
- Create: `backend/prisma/migrations/20260607000000_add_backup_codes/migration.sql`

- Modify: `backend/src/users/users.controller.ts`（`POST me/backup-code` と `POST me/restore` 追加のみ）
- Modify: `backend/src/users/users.module.ts`（BackupCodesRepository を providers に追加）
- Modify: `frontend/src/lib/api/backup-code.ts`（URL を `/users/me/backup-code` / `/users/me/restore` に更新）

### 設計

**バックアップコード形式:** `ALRG-XXXX-XXXX`（X = ランダム英数字大文字、計8文字）  
**保存:** コードを SHA-256 でハッシュ化して保存（コード本体はレスポンスのみ）  
**有効期限:** 発行から 30 日  
**制約:** 1ユーザーにつき有効コードは 1 件のみ（再発行時は古いコードを無効化）  
**restore の動作:** コードからコードの発行元ユーザーを特定し、その allergies を現在ログイン中のユーザーにコピーする。コードを使用済みにする。

### コントローラーの方針

既存の `@Controller('users/me')` を **そのまま維持** する。バックアップコード操作も「自分の」データへの操作なので `/users/me/` 下が一貫性あり。

- `@Post('backup-code')` → `POST /users/me/backup-code`
- `@Post('restore')` → `POST /users/me/restore`

frontend の `backup-code.ts` が現在 `/users/backup-code` / `/users/restore` を呼んでいるため、そちらも合わせて更新する。

- [ ] **Step 3: BackupCodesRepository を作成する**

新規ファイル `backend/src/users/backup-codes.repository.ts`:

```typescript
import { Injectable } from '@nestjs/common'
import { createHash, randomBytes } from 'crypto'
import { PrismaService } from '../prisma/prisma.service'

const CODE_TTL_DAYS = 30

const generateCode = (): string => {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'  // 紛らわしい文字 O, 0, I, 1 を除外
  const part = (len: number) =>
    Array.from({ length: len }, () => chars[randomBytes(1)[0] % chars.length]).join('')
  return `ALRG-${part(4)}-${part(4)}`
}

const hashCode = (code: string): string =>
  createHash('sha256').update(code).digest('hex')

export type IssuedBackupCode = {
  code: string
  expires_at: Date
}

@Injectable()
export class BackupCodesRepository {
  constructor(private readonly prisma: PrismaService) {}

  async issue(userId: string): Promise<IssuedBackupCode> {
    const code = generateCode()
    const codeHash = hashCode(code)
    const expiresAt = new Date(Date.now() + CODE_TTL_DAYS * 24 * 60 * 60 * 1000)

    // 既存の未使用コードを全て無効化（used_at をセット）してから新規発行
    await this.prisma.backupCode.updateMany({
      where: { userId, usedAt: null },
      data: { usedAt: new Date() },
    })

    await this.prisma.backupCode.create({
      data: { userId, codeHash, expiresAt },
    })

    return { code, expires_at: expiresAt }
  }

  async findValidByCode(code: string): Promise<{ id: string; userId: string } | null> {
    const codeHash = hashCode(code)
    const record = await this.prisma.backupCode.findFirst({
      where: {
        codeHash,
        usedAt: null,
        expiresAt: { gt: new Date() },
      },
      select: { id: true, userId: true },
    })
    return record ?? null
  }

  async markUsed(id: string): Promise<void> {
    await this.prisma.backupCode.update({
      where: { id },
      data: { usedAt: new Date() },
    })
  }
}
```

```typescript
import { BadRequestException, Injectable, Logger, NotFoundException } from '@nestjs/common'
import { BackupCodesRepository } from './backup-codes.repository'

// コンストラクタ引数を追加:
constructor(
  private readonly usersRepository: UsersRepository,
  private readonly backupCodesRepository: BackupCodesRepository,
) {}

async issueBackupCode(userId: string): Promise<{ code: string; expires_at: string }> {
  const user = await this.usersRepository.findById(userId)
  if (!user) throw new NotFoundException('ユーザーが見つかりません')
  const result = await this.backupCodesRepository.issue(userId)
  return { code: result.code, expires_at: result.expires_at.toISOString() }
}

async restoreFromCode(requestingUserId: string, code: string): Promise<void> {
  const record = await this.backupCodesRepository.findValidByCode(code)
  if (!record) throw new BadRequestException('code_invalid')

  // コードの発行元ユーザーのアレルギー設定を取得
  const sourceUser = await this.usersRepository.findById(record.userId)
  if (!sourceUser) throw new BadRequestException('code_invalid')

  // 現在ユーザーにアレルギー設定をコピー
  await this.usersRepository.update(requestingUserId, { allergies: sourceUser.allergies })

  // コードを使用済みにする
  await this.backupCodesRepository.markUsed(record.id)

  this.logger.log('バックアップコードで引継ぎ完了', {
    requestingUserId,
    sourceUserId: record.userId,
  })
}
```

```typescript
  return res.json() as Promise<IssueBackupCodeResponse>
}

// POST /users/restore → POST /users/me/restore
export const restoreFromCode = async (
  code: string,
): Promise<RestoreFromCodeResponse> => {
  const baseUrl = process.env.NEXT_PUBLIC_API_BASE_URL ?? ''
  const { createClient } = await import('@/lib/supabase/client')
  const supabase = createClient()
  const { data: { session } } = await supabase.auth.getSession()
  const token = session?.access_token

  const headers: HeadersInit = {
    'Content-Type': 'application/json',
    ...(token ? { Authorization: `Bearer ${token}` } : {}),
  }

  const res = await fetch(`${baseUrl}/users/me/restore`, {  // ← /users/restore から変更
    method: 'POST',
    headers,
    body: JSON.stringify({ code }),
  })
  if (res.status === 400) {
    const body = (await res.json()) as RestoreFromCodeError
    throw body
  }
  if (!res.ok) {
    throw new Error(`POST /users/me/restore failed: ${res.status}`)
  }
  return res.json() as Promise<RestoreFromCodeResponse>
}
```
